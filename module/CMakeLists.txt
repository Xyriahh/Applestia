cmake_minimum_required(VERSION 3.25)
project(applestia-glass VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.11 REQUIRED COMPONENTS Quick QuickPrivate Qml WaylandClient WaylandClientPrivate Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
include(GNUInstallDirs)
set(GLASS_PROTOCOL "${CMAKE_CURRENT_SOURCE_DIR}/../plugin/protocols/applestia-glass-shapes-v1.xml" CACHE FILEPATH "Shared compositor protocol")
set(GLASS_QML_INSTALL_DIR "${CMAKE_INSTALL_LIBDIR}/qt6/qml" CACHE STRING "QML import root (relative to prefix)")
qt_standard_project_setup(REQUIRES 6.11)
qt_add_qml_module(applestiaglass URI Applestia.Glass VERSION 1.0
    OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml/Applestia/Glass"
    DEPENDENCIES QtQuick
    SOURCES liquidglass.h liquidglass.cpp)
qt6_generate_wayland_protocol_client_sources(applestiaglass FILES "${GLASS_PROTOCOL}")
target_link_libraries(applestiaglass PRIVATE Qt6::QuickPrivate Qt6::WaylandClientPrivate PkgConfig::WAYLAND)
install(TARGETS applestiaglass applestiaglassplugin LIBRARY DESTINATION "${GLASS_QML_INSTALL_DIR}/Applestia/Glass")
install(FILES "${CMAKE_BINARY_DIR}/qml/Applestia/Glass/qmldir" "${CMAKE_BINARY_DIR}/qml/Applestia/Glass/applestiaglass.qmltypes" DESTINATION "${GLASS_QML_INSTALL_DIR}/Applestia/Glass")
include(CTest)
if(BUILD_TESTING)
target_compile_definitions(applestiaglass PRIVATE APPLESTIA_GLASS_TEST_HOOKS)
qt_add_executable(glass-tests tests.cpp)
target_link_libraries(glass-tests PRIVATE applestiaglass Qt6::Quick Qt6::Test PkgConfig::WAYLAND)
target_include_directories(glass-tests PRIVATE "${CMAKE_BINARY_DIR}")
add_test(NAME glass-logic COMMAND glass-tests)
set_tests_properties(glass-logic PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
qt_add_executable(glass-import-test import-test.cpp)
target_link_libraries(glass-import-test PRIVATE Qt6::Quick Qt6::Qml)
add_test(NAME glass-import COMMAND glass-import-test "${CMAKE_BINARY_DIR}/qml")
set_tests_properties(glass-import PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
qt_add_executable(glass-shell-tests shell-tests.cpp)
target_link_libraries(glass-shell-tests PRIVATE Qt6::Quick Qt6::Qml Qt6::Test)
target_compile_definitions(glass-shell-tests PRIVATE SHELL_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/../shell" QML_BUILD="${CMAKE_BINARY_DIR}/qml")
add_test(NAME glass-shell COMMAND glass-shell-tests)
set_tests_properties(glass-shell PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
